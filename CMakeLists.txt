cmake_minimum_required(VERSION 3.20)
project(Snipper VERSION 1.0.1 LANGUAGES CXX RC)
add_executable(Snipper WIN32 src/main.cpp src/model.cpp src/graphics.cpp resources/app.rc)
target_compile_features(Snipper PRIVATE cxx_std_20)
target_compile_definitions(Snipper PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(Snipper PRIVATE resources)
target_link_libraries(Snipper PRIVATE d2d1 dwrite windowscodecs ole32 user32 gdi32 comdlg32 comctl32 shell32 advapi32 dwmapi)
if(MSVC)
  target_compile_options(Snipper PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET Snipper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(Snipper PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(Snipper PRIVATE -municode -static -s)
endif()
enable_testing()
add_test(NAME native_self_test COMMAND Snipper --self-test)
add_executable(clipboard_test tests/clipboard.cpp src/graphics.cpp src/model.cpp)
target_compile_features(clipboard_test PRIVATE cxx_std_20)
target_compile_definitions(clipboard_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(clipboard_test PRIVATE src)
target_link_libraries(clipboard_test PRIVATE d2d1 dwrite windowscodecs ole32 user32 gdi32)
if(MSVC)
  target_compile_options(clipboard_test PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET clipboard_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(clipboard_test PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(clipboard_test PRIVATE -municode -static -s)
endif()
add_test(NAME isolated_clipboard_test COMMAND clipboard_test)
